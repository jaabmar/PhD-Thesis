\chapter*{Publications}

This thesis is based on material covered in the following peer-reviewed publications.

\begin{itemize}[leftmargin=*]

  \item \textbf{Chapter~\ref{chap:confounding-test}.}
  Piersilvio De Bartolomeis\equalcontributionmark,
  Javier Abad\equalcontributionmark,
  Konstantin Donhauser, and Fanny Yang.
  ``Hidden yet quantifiable: A lower bound for confounding
  strength using randomized trials.''
  In \emph{Proceedings of the 27th International Conference
  on Artificial Intelligence and Statistics (AISTATS)},
  \emph{Proceedings of Machine Learning Research},
  vol.~238, pp.~1045--1053, 2024.
  \url{https://proceedings.mlr.press/v238/de-bartolomeis24a.html}.

  \item \textbf{Chapter~\ref{chap:privpgd}.}
  Konstantin Donhauser\equalcontributionmark,
  Javier Abad\equalcontributionmark,
  Neha Hulkund, and Fanny Yang.
  ``Privacy-Preserving Data Release Leveraging Optimal
  Transport and Particle Gradient Descent.''
  In \emph{Proceedings of the 41st International Conference
  on Machine Learning (ICML)},
  \emph{Proceedings of Machine Learning Research},
  vol.~235, pp.~11453--11473, 2024.
  \url{https://proceedings.mlr.press/v235/donhauser24a.html}.

  \item \textbf{Chapter~\ref{chap:cpfuse}.}
  Javier Abad, Konstantin Donhauser,
  Francesco Pinto\textsuperscript{\dag},
  and Fanny Yang\textsuperscript{\dag}.
  ``Copyright-Protected Language Generation via Adaptive Model Fusion.''
  In \emph{Proceedings of the 13th International Conference
  on Learning Representations (ICLR)}, 2025.
  \url{https://proceedings.iclr.cc/paper_files/paper/2025/hash/01953220e4becc6ac1078c24c1f8d3a7-Abstract-Conference.html}.

\end{itemize}

\noindent\equalcontributionnote\quad{\footnotesize\textsuperscript{\dag}\,Equal supervision.}

\bigskip
\noindent\textbf{Author contributions.} In Chapter~\ref{chap:confounding-test}, I co-developed the problem formulation and lower-bound testing procedure, and led the experimental design and evaluation across the synthetic, semi-synthetic, and real-world experiments. In Chapter~\ref{chap:privpgd}, I co-designed PrivPGD, implemented the particle gradient descent method and the marginal-based baselines, and carried out the benchmark comparison across tabular datasets. I led the work in Chapter~\ref{chap:cpfuse}, where I designed and implemented CP-Fuse and conducted the text and code experiments, prefix-extraction attacks and ablations.
